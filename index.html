/* Nuanced polarity + chroma-weighted shift */ /* High contrast adds chroma boost (+0.162) */ @function --auto-lightness( --bg <color>, --level <number>, --fg-c <number> ) returns <color> { /* effectiveShift = level × (1 + bgChroma) */ /* Polarity: if C > 0.15 AND L ≥ 0.75 → dark text */ result: if( c(from var(--bg)) > 0.15 and l(from var(--bg)) >= 0.75: oklch( from var(--bg) calc(l - var(--level) * (1 + c)) var(--fg-c) h ); else: oklch( from var(--bg) calc(l + var(--level) * (1 + c)) var(--fg-c) h ) ); } /* Usage */ .card { --bg: oklch(0.700 0.150 250); background: var(--bg); color: --auto-lightness( var(--bg), 0.35, 0.000 ); }
